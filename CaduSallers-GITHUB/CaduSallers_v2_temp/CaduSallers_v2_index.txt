const carts = new Map();
const orders = new Map();
const tickets = new Map();

const BRAND = 0x5865F2;
const SUCCESS = 0x57F287;
const WARNING = 0xFEE75C;
const DANGER = 0xED4245;
const DARK = 0x2B2D31;

function money(value) {
    return Number(value).toLocaleString('pt-BR', {
        style: 'currency',
        currency: 'BRL'
    });
}

function isAdmin(interaction) {
    return Boolean(
        interaction.memberPermissions?.has(PermissionFlagsBits.Administrator)
    );
}

function getCart(userId) {
    if (!carts.has(userId)) carts.set(userId, []);
    return carts.get(userId);
}

function getOrders(userId) {
    if (!orders.has(userId)) orders.set(userId, []);
    return orders.get(userId);
}

function totalStock() {
    return products.reduce((sum, product) => sum + product.stock, 0);
}

function totalOrders() {
    let total = 0;
    for (const list of orders.values()) total += list.length;
    return total;
}

function totalRevenue() {
    let total = 0;
    for (const list of orders.values()) {
        for (const order of list) {
            if (order.status === 'Pago') total += order.total;
        }
    }
    return total;
}

    const text = description ||
        'Central de gerenciamento da sua operação de vendas.\n\n' +
        'Escolha uma seção no menu abaixo para continuar.';

    return new EmbedBuilder()
        .setColor(BRAND)
        .setTitle(title)
        .setDescription(text)
        .addFields(
            { name: 'Produtos', value: String(products.length), inline: true },
            { name: 'Estoque', value: String(totalStock()), inline: true },
            { name: 'Pedidos', value: String(totalOrders()), inline: true },
            { name: 'Faturamento pago', value: money(totalRevenue()), inline: true },
            { name: 'Tickets', value: String(tickets.size), inline: true },
            { name: 'Categorias', value: String(new Set(products.map(p => p.category)).size), inline: true }

        .setFooter({ text: 'CaduSallers • Painel' })
        .setTimestamp();
}

function storeEmbed() {
    const lines = products.map((p) =>
        `**${p.name}**\n${p.description}\n${money(p.price)} • ${p.stock} em estoque`
    );

    return new EmbedBuilder()
        .setColor(BRAND)
        .setTitle('Loja')
        .setDescription(lines.length ? lines.join('\n\n') : 'Nenhum produto cadastrado.')
        .setFooter({ text: 'Selecione um produto abaixo para abrir os detalhes.' });
}

function productMenu() {
    const options = products.slice(0, 25).map((p) => ({
        label: p.name.slice(0, 100),
        description: `${money(p.price)} • ${p.stock} em estoque`.slice(0, 100),
        value: p.id
    }));

    return new ActionRowBuilder().addComponents(
        new StringSelectMenuBuilder()
            .setCustomId('product_select')
            .setPlaceholder('Selecionar produto')
            .addOptions(options)
    );
}

function productDetail(product) {
    return new EmbedBuilder()
        .setColor(BRAND)
        .setTitle(product.name)
        .setDescription(product.description)
        .addFields(
            { name: 'Preço', value: money(product.price), inline: true },
            { name: 'Estoque', value: String(product.stock), inline: true },
            { name: 'Categoria', value: product.category, inline: true }
        )
        .setFooter({ text: 'CaduSallers • Produto' });
}

function productActions(productId) {
    return new ActionRowBuilder().addComponents(
        new ButtonBuilder()
            .setCustomId(`add:${productId}`)
            .setLabel('Adicionar ao carrinho')
            .setStyle(ButtonStyle.Success),
        new ButtonBuilder()
            .setCustomId(`buy:${productId}`)
            .setLabel('Comprar agora')
            .setStyle(ButtonStyle.Primary),
        new ButtonBuilder()
            .setCustomId('back_store')
            .setLabel('Voltar')
            .setStyle(ButtonStyle.Secondary)
    );
}

function cartEmbed(userId) {
    const cart = getCart(userId);

    if (!cart.length) {
        return new EmbedBuilder()
            .setColor(DARK)
            .setTitle('Carrinho')
            .setDescription('Seu carrinho está vazio.')
            .setFooter({ text: 'Volte à loja para adicionar produtos.' });
    }

    let total = 0;
    const lines = [];

    for (const item of cart) {
        const product = products.find(p => p.id === item.productId);
        if (!product) continue;
        const subtotal = product.price * item.quantity;
        total += subtotal;
        lines.push(`**${product.name}** × ${item.quantity} — ${money(subtotal)}`);
    }

    return new EmbedBuilder()
        .setColor(BRAND)
        .setTitle('Carrinho')
        .setDescription(lines.join('\n') || 'Carrinho vazio.')
        .addFields({ name: 'Total', value: `**${money(total)}**` })
        .setFooter({ text: 'Revise os itens antes do checkout.' });
}

function cartActions(userId) {
    const hasItems = getCart(userId).length > 0;

    return new ActionRowBuilder().addComponents(
        new ButtonBuilder()
            .setCustomId('checkout')
            .setLabel('Finalizar compra')
            .setStyle(ButtonStyle.Success)
            .setDisabled(!hasItems),
        new ButtonBuilder()
            .setCustomId('clear_cart')
            .setLabel('Limpar carrinho')
            .setStyle(ButtonStyle.Danger)
            .setDisabled(!hasItems),
        new ButtonBuilder()
            .setCustomId('back_store')
            .setLabel('Voltar à loja')
            .setStyle(ButtonStyle.Secondary)
    );
}

client.on('interactionCreate', async (interaction) => {
    try {
        if (interaction.isChatInputCommand()) {
            if (interaction.commandName === 'panel') {
                await interaction.reply({
                    embeds: [panelEmbed()],
                    components: panelComponents()
                });
                return;
            }

            if (interaction.commandName === 'loja') {
                await interaction.reply({
                    embeds: [storeEmbed()],
                    components: [productMenu(), new ActionRowBuilder().addComponents(
                        new ButtonBuilder()
                            .setCustomId('open_cart')
                            .setLabel('Carrinho')
                            .setStyle(ButtonStyle.Secondary),
                        new ButtonBuilder()
                            .setCustomId('panel_home')
                            .setLabel('Painel')
                            .setStyle(ButtonStyle.Secondary)
                    )]
                });
                return;
            }

            if (interaction.commandName === 'vercompras') {
                await interaction.reply({
                    embeds: [purchasesEmbed(interaction.user.id)],
                    ephemeral: true
                });
                return;
            }

            if (interaction.commandName === 'tools') {
                if (!isAdmin(interaction)) {
                    await interaction.reply({ content: 'Você não possui permissão para usar este painel.', ephemeral: true });
                    return;
                }

                await interaction.reply({
                    embeds: [new EmbedBuilder()
                        .setColor(WARNING)
                        .setTitle('Ferramentas')
                        .setDescription(
                            'Ferramentas administrativas do CaduSallers.\n\n' +
                            'A estrutura está pronta para receber módulos como separadores, utilidades, banners e outras ferramentas.'
                        )],
                    components: [new ActionRowBuilder().addComponents(
                        new ButtonBuilder().setCustomId('panel_home').setLabel('Painel').setStyle(ButtonStyle.Secondary)
                    )]
                });
                return;
            }

            if (interaction.commandName === 'criar_produto') {
                if (!isAdmin(interaction)) {
                    await interaction.reply({ content: 'Você não possui permissão para isso.', ephemeral: true });
                    return;
                }

                const next = products.length + 1;
                products.push({
                    id: `produto-${next}`,
                    name: `Novo Produto ${next}`,
                    description: 'Produto criado pelo comando de demonstração.',
                    price: 25.00,
                    stock: 0,
                    category: 'Geral'
                });

                await interaction.reply({
                    content: `Produto **Novo Produto ${next}** criado.`,
                    ephemeral: true
                });
                return;
            }

            if (interaction.commandName === 'setup') {
                if (!isAdmin(interaction)) {
                    await interaction.reply({ content: 'Você não possui permissão para usar o setup.', ephemeral: true });
                    return;
                }

                if (!interaction.channel || interaction.channel.type !== ChannelType.GuildText) {
                    await interaction.reply({ content: 'Use este comando em um canal de texto do servidor.', ephemeral: true });
                    return;
                }

                await interaction.channel.send({
                    embeds: [panelEmbed()],
                    components: panelComponents()
                });

                await interaction.reply({ content: 'Painel publicado neste canal.', ephemeral: true });
                return;
            }
        }

        if (interaction.isStringSelectMenu()) {
            if (interaction.customId === 'panel_section') {
                const key = interaction.values[0];

                if (key === 'store') {
                    await interaction.update({
                        embeds: [storeEmbed()],
                        components: [
                            productMenu(),
                            new ActionRowBuilder().addComponents(
                                new ButtonBuilder().setCustomId('open_cart').setLabel('Carrinho').setStyle(ButtonStyle.Secondary),
                                new ButtonBuilder().setCustomId('panel_home').setLabel('Painel').setStyle(ButtonStyle.Secondary)
                            )
                        ]
                    });
                    return;
                }

                if (key === 'purchases') {
                    await interaction.update({
                        embeds: [purchasesEmbed(interaction.user.id)],
                        components: [
                            new ActionRowBuilder().addComponents(
                                new ButtonBuilder().setCustomId('panel_home').setLabel('Voltar ao painel').setStyle(ButtonStyle.Secondary)
                            )
                        ]
                    });
                    return;
                }

                if (key === 'metrics') {
                    await interaction.update({
                        embeds: [metricsEmbed()],
                        components: [panelButtons()]
                    });
                    return;
                }

                if (key === 'tools' && !isAdmin(interaction)) {
                    await interaction.reply({ content: 'Você não possui permissão para acessar as ferramentas.', ephemeral: true });
                    return;
                }

                if (sectionData[key]) {
                    await interaction.update({
                        embeds: [sectionEmbed(key)],
                        components: [panelMenu(), sectionButtons()]
                    });
                    return;
                }
            }

            if (interaction.customId === 'product_select') {
                const product = products.find(p => p.id === interaction.values[0]);

                if (!product) {
                    await interaction.reply({ content: 'Produto não encontrado.', ephemeral: true });
                    return;
                }

                await interaction.update({
                    embeds: [productDetail(product)],
                    components: [productActions(product.id)]
                });
                return;
            }
        }

        if (interaction.isButton()) {
            const id = interaction.customId;

            if (id === 'panel_home' || id === 'panel_refresh') {
                await interaction.update({
                    embeds: [panelEmbed()],
                    components: panelComponents()
                });
                return;
            }

            if (id === 'panel_close') {
                await interaction.update({
                    content: 'Painel fechado.',
                    embeds: [],
                    components: []
                });
                return;
            }

            if (id === 'back_store') {
                await interaction.update({
                    embeds: [storeEmbed()],
                    components: [productMenu(), new ActionRowBuilder().addComponents(
                        new ButtonBuilder().setCustomId('open_cart').setLabel('Carrinho').setStyle(ButtonStyle.Secondary),
                        new ButtonBuilder().setCustomId('panel_home').setLabel('Painel').setStyle(ButtonStyle.Secondary)
                    )]
                });
                return;
            }

            if (id === 'open_cart') {
                await interaction.update({
                    embeds: [cartEmbed(interaction.user.id)],
                    components: [cartActions(interaction.user.id)]
                });
                return;
            }

            if (id.startsWith('add:')) {
                const product = products.find(p => p.id === id.slice(4));

                if (!product) {
                    await interaction.reply({ content: 'Produto não encontrado.', ephemeral: true });
                    return;
                }

                if (product.stock <= 0) {
                    await interaction.reply({ content: 'Este produto está sem estoque.', ephemeral: true });
                    return;
                }

                const cart = getCart(interaction.user.id);
                const existing = cart.find(item => item.productId === product.id);

                if (existing) {
                    if (existing.quantity >= product.stock) {
                        await interaction.reply({ content: 'Você já adicionou o máximo disponível em estoque.', ephemeral: true });
                        return;
                    }
                    existing.quantity += 1;
                } else {
                    cart.push({ productId: product.id, quantity: 1 });
                }

                await interaction.reply({ content: `${product.name} adicionado ao carrinho.`, ephemeral: true });
                return;
            }

            if (id.startsWith('buy:')) {
                const product = products.find(p => p.id === id.slice(4));

                if (!product) {
                    await interaction.reply({ content: 'Produto não encontrado.', ephemeral: true });
                    return;
                }

                const order = {
                    id: `PED-${Date.now().toString(36).toUpperCase()}`,
                    total: product.price,
                    status: 'Pendente',
                    items: [{ productId: product.id, quantity: 1 }]
                };

                getOrders(interaction.user.id).push(order);

                await interaction.reply({
                    embeds: [new EmbedBuilder()
                        .setColor(WARNING)
                        .setTitle('Checkout')
                        .setDescription(
                            `Pedido **${order.id}** criado.\n\n` +
                            `Produto: **${product.name}**\n` +
                            `Total: **${money(order.total)}**\n\n` +
                            'A etapa de pagamento será conectada ao gateway na próxima versão.'
                        )],
                    ephemeral: true
                });
                return;
            }

            if (id === 'clear_cart') {
                carts.set(interaction.user.id, []);
                await interaction.update({
                    embeds: [cartEmbed(interaction.user.id)],
                    components: [cartActions(interaction.user.id)]
                });
                return;
            }

            if (id === 'checkout') {
                const cart = getCart(interaction.user.id);

                if (!cart.length) {
                    await interaction.reply({ content: 'Seu carrinho está vazio.', ephemeral: true });
                    return;
                }

                let total = 0;
                for (const item of cart) {
                    const product = products.find(p => p.id === item.productId);
                    if (product) total += product.price * item.quantity;
                }

                const order = {
                    id: `PED-${Date.now().toString(36).toUpperCase()}`,
                    total,
                    status: 'Pendente',
                    items: [...cart]
                };

                getOrders(interaction.user.id).push(order);
                carts.set(interaction.user.id, []);

                await interaction.update({
                    content: null,
                    embeds: [new EmbedBuilder()
                        .setColor(WARNING)
                        .setTitle('Pedido criado')
                        .setDescription(
                            `Pedido: **${order.id}**\n` +
                            `Total: **${money(order.total)}**\n` +
                            'Status: **Pendente**\n\n' +
                            'O pagamento Pix/cripto e a entrega automática serão conectados ao checkout em seguida.'
                        )],
                    components: [new ActionRowBuilder().addComponents(
                        new ButtonBuilder().setCustomId('panel_home').setLabel('Painel').setStyle(ButtonStyle.Secondary),
                        new ButtonBuilder().setCustomId('panel_refresh').setLabel('Atualizar').setStyle(ButtonStyle.Secondary)
                    )]
                });
                return;
            }

            if (id === 'section_action_1') {
                await interaction.reply({ content: 'O módulo está criado e será ligado às operações persistentes nas próximas etapas.', ephemeral: true });
            }
        }
    } catch (error) {
        console.error('Erro na interação:', error);

        try {
            if (interaction.replied || interaction.deferred) {
                await interaction.followUp({ content: 'Ocorreu um erro ao processar esta ação.', ephemeral: true });
            } else {
                await interaction.reply({ content: 'Ocorreu um erro ao processar esta ação.', ephemeral: true });
            }
        } catch {
            // Evita erro secundário no tratamento de falhas.
        }
    }
});

client.once('ready', async () => {
    console.log('');
    console.log('========================================');
    console.log(`Cadu BOT conectado como ${client.user.tag}`);
    console.log(`Servidores: ${client.guilds.cache.size}`);
    console.log('========================================');

    const rest = new REST({ version: '10' }).setToken(token);

    try {
        for (const guild of client.guilds.cache.values()) {
            await rest.put(
                Routes.applicationGuildCommands(client.user.id, guild.id),
                { body: commands.map(command => command.toJSON()) }
            );
            console.log(`Comandos registrados em: ${guild.name}`);
        }
    } catch (error) {
        console.error('Erro ao registrar comandos:', error);
    }
});

client.on('error', error => {
    console.error('Erro do Discord:', error);
});

process.on('unhandledRejection', error => {
    console.error('Unhandled rejection:', error);
});

client.login(token);
